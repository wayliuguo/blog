demo · 整体运行（官网 CDN 用法）

页面加载即按官网 CDN 的方式运行：createRoot(document.getElementById('app')).render(h(App)) ——App 是一个函数组件，内部用 useState 持有 message / count / todos，两个 useEffect 分别对应"挂载"与"[message] 依赖"。左栏是组件源码（h 就是 createElement，即 JSX 的运行时形态），右栏渲染区就是这个组件；下面的页面层按钮负责触发各条链路。

组件源码（与官网 CDN 示例同构）

const M = window.MiniReact

function App({ onReady }) {
  const [message, setMessage] = M.useState('Hello React!')
  const [count, setCount] = M.useState(0)
  const [todos, setTodos] = M.useState([
    { id: 1, text: 'a' },
    { id: 2, text: 'b' },
    { id: 3, text: 'c' },
    { id: 4, text: 'd' },
    { id: 5, text: 'e' }
  ])
  M.useEffect(() => {
    onReady({ setMessage, setCount, setTodos })
  }, [])
  M.useEffect(() => {
    lifeMark.append(`\nuseEffect([message]) 触发：${message}`)
    return () => lifeMark.append(`\n上一次 effect 清理执行`)
  }, [message])
  return M.h('div', null,
    M.h('p', null, message),
    M.h('p', null, 'count = ', count),
    M.h('button', { onClick: () => setCount(count + 1) }, 'count + 1'),
    M.h('ul', null, todos.map(t =>
      M.h('li', { key: t.id }, '#', t.id, ' ', t.text)
    ))
  )
}

M.createRoot(document.getElementById('app'))
  .render(M.h(App, { onReady: api => (appApi = api) }))

断点标注（按链路顺序，对应《React 原理》五章）
  1. createRoot().render(h(App))：建 HostRoot fiber（stateNode = 容器）→ scheduleRootUpdate（断点：scheduler.js 的 scheduleCallback）→ MessageChannel 宏任务起跑
  2. performRootWork：开新工作栈（createWorkInProgress 双缓冲）→ workLoop 逐 fiber 推进（断点：performUnitOfWork）→ App 的 beginWork 走 renderWithHooks（断点：hooks 链表建立）
  3. completeWork：宿主 fiber 创建游离 DOM 实例并铺属性； commitRoot（断点：commitPlacement + getHostSibling）→ flushPassiveEffects：useEffect([]) 先执行 → onReady 递出 setState
  4. 改状态：dispatchSetState 只记 action + 排调度（断点：看 hook.queue）→ 下一次 render 从根重算 → reconcileChildren（keyed diff：头同步 → 尾同步 → 乱序区 lastPlacedIndex 判移动）→ commitMutationEffects（先删后插）→ flushPassiveEffects 先 destroy 再 create
  5. unmount()：commitDeletion（断点：同步执行 useEffect destroy）→ 摘除 DOM；重新加载页面即可重看挂载
