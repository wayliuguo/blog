## [中级] 用户注册

### 密码加密

```bash
npm install bcrypt
npm install -D @types/bcrypt
```

```typescript
import * as bcrypt from 'bcrypt'

@Injectable()
export class AuthService {
    async register(dto: RegisterDto) {
        // 1. 检查用户是否已存在
        const existing = await this.userRepository.findOneBy({ email: dto.email })
        if (existing) {
            throw new ConflictException('邮箱已被注册')
        }

        // 2. 密码加密
        const salt = await bcrypt.genSalt(10)  // 生成盐值，10 是加密轮数
        const hashedPassword = await bcrypt.hash(dto.password, salt)

        // 3. 保存用户
        const user = this.userRepository.create({
            ...dto,
            password: hashedPassword
        })
        return this.userRepository.save(user)
    }
}
```

### 数据校验

```typescript
export class RegisterDto {
    @IsEmail()
    email: string

    @MinLength(6)
    @MaxLength(20)
    @Matches(/(?=.*[a-z])(?=.*[A-Z])(?=.*\d)/, {
        message: '密码必须包含大小写字母和数字'
    })
    password: string

    @MinLength(2)
    @MaxLength(50)
    name: string
}
```

## [中级] JWT 登录

### 安装

```bash
npm install @nestjs/jwt @nestjs/passport passport passport-jwt
npm install -D @types/passport-jwt
```

### 签发 JWT

```typescript
import { JwtService } from '@nestjs/jwt'

@Injectable()
export class AuthService {
    constructor(
        private jwtService: JwtService,
        private userRepository: Repository<User>
    ) {}

    async login(dto: LoginDto) {
        // 1. 查找用户
        const user = await this.userRepository.findOneBy({ email: dto.email })
        if (!user) {
            throw new UnauthorizedException('邮箱或密码错误')
        }

        // 2. 验证密码
        const isMatch = await bcrypt.compare(dto.password, user.password)
        if (!isMatch) {
            throw new UnauthorizedException('邮箱或密码错误')
        }

        // 3. 签发 JWT
        const payload = { sub: user.id, email: user.email }
        return {
            access_token: this.jwtService.sign(payload),  // 默认 HS256
            user: { id: user.id, name: user.name, email: user.email }
        }
    }
}
```

### JWT 验证 Guard

```typescript
import { Injectable } from '@nestjs/common'
import { PassportStrategy } from '@nestjs/passport'
import { ExtractJwt, Strategy } from 'passport-jwt'

@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy) {
    constructor() {
        super({
            jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
            ignoreExpiration: false,
            secretOrKey: 'your-secret-key'
        })
    }

    async validate(payload: { sub: number; email: string }) {
        // 返回的数据会挂载到 request.user 上
        return { userId: payload.sub, email: payload.email }
    }
}

// 使用
@Controller('users')
export class UserController {
    @UseGuards(AuthGuard('jwt'))  // 需要登录才能访问
    @Get('profile')
    getProfile(@Req() req: any) {
        return req.user  // 从 JWT 解析出来的用户信息
    }
}
```

### @Public() 跳过验证

```typescript
import { SetMetadata } from '@nestjs/common'

export const IS_PUBLIC_KEY = 'isPublic'
export const Public = () => SetMetadata(IS_PUBLIC_KEY, true)

// 全局 Guard 中判断
@Injectable()
export class JwtAuthGuard extends AuthGuard('jwt') {
    constructor(private reflector: Reflector) {
        super()
    }

    canActivate(context: ExecutionContext) {
        const isPublic = this.reflector.getAllAndOverride<boolean>(IS_PUBLIC_KEY, [
            context.getHandler(),
            context.getClass()
        ])
        if (isPublic) return true  // 跳过验证
        return super.canActivate(context)
    }
}

// 使用
@Public()
@Post('login')
login(@Body() dto: LoginDto) {
    return this.authService.login(dto)
}
```

## [中级] Refresh Token 机制

### 双 Token 策略

```typescript
// Access Token：短期有效（2小时），用于访问资源
// Refresh Token：长期有效（7天），用于获取新的 Access Token

@Injectable()
export class AuthService {
    async login(dto: LoginDto) {
        const user = await this.validateUser(dto)
        const payload = { sub: user.id, email: user.email }

        return {
            access_token: this.jwtService.sign(payload, { expiresIn: '2h' }),
            refresh_token: this.jwtService.sign(payload, { expiresIn: '7d' }),
        }
    }

    async refreshToken(refreshToken: string) {
        try {
            const payload = this.jwtService.verify(refreshToken)
            const newPayload = { sub: payload.sub, email: payload.email }

            return {
                access_token: this.jwtService.sign(newPayload, { expiresIn: '2h' }),
            }
        } catch {
            throw new UnauthorizedException('Refresh Token 已过期，请重新登录')
        }
    }
}
```

## [高级] 微信小程序登录

```typescript
@Injectable()
export class WechatAuthService {
    constructor(
        private userRepository: Repository<User>,
        private jwtService: JwtService,
    ) {}

    async wechatLogin(code: string) {
        // 1. 用 code 向微信服务器换取 session_key 和 openid
        const response = await axios.get('https://api.weixin.qq.com/sns/jscode2session', {
            params: {
                appid: 'your-app-id',
                secret: 'your-app-secret',
                js_code: code,
                grant_type: 'authorization_code'
            }
        })

        const { openid, session_key } = response.data

        // 2. 查找或创建用户
        let user = await this.userRepository.findOneBy({ wechatOpenId: openid })
        if (!user) {
            user = this.userRepository.create({ wechatOpenId: openid })
            user = await this.userRepository.save(user)
        }

        // 3. 签发 JWT
        const payload = { sub: user.id, openid: openid }
        return {
            access_token: this.jwtService.sign(payload)
        }
    }
}
```

## 面试题

### Q1: JWT 的结构是什么？为什么不需要在服务端存储 Session？

JWT 由 Header（算法）、Payload（数据）、Signature（签名）三部分组成，用 `.` 分隔。服务端不需要存储 Session，因为 JWT 本身包含了用户信息和签名，服务端只需验证签名即可信任其内容。

### Q2: 为什么要使用 Refresh Token？

Access Token 有效期短（2小时），即使泄露损失也有限。Refresh Token 有效期长（7天），用于无感刷新 Access Token，避免用户频繁重新登录。两者结合实现安全性和用户体验的平衡。
